var unitsSoldAnHour = function(salesModifier, price, quality, marketSaturation, retailModeling) {
  var a = Math.max(marketSaturation - 0.24 * quality, 0.1);
  var i = timeModeling(retailModeling, a, 100, price);
  return 360000 / (i - i * salesModifier / 100);
};

var profitPerUnit = function(salesModifier, price, quality, marketSaturation, retailModeling, administrationOverhead, storeBaseSalary) {

var unitsAnHour = function() {
  var e = r.state,
      size = e.size,
      productionModifier = e.productionModifier,
      abundance = e.abundance,
      resource = r.props.resource,
      producedAnHour = resource.producedAnHour,
      db_letter = resource.db_letter,
      isAbundanceResource = ['m', 'o', 'c'].indexOf(db_letter) !== -1,
      effectiveRate = isAbundanceResource ? (producedAnHour * abundance / 100) : producedAnHour;

  return size * effectiveRate / (1 - productionModifier / 100);
};

// 2019 行政管理对工人工资加成损耗公式
